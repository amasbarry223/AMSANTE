    52→  laboratoire: "#D9A21B",

    60→const serviceIcons: Record<GuaranteeKey, LucideIcon> = {

    75→
    76→const pillBase =
    77→  "inline-flex min-h-11 items-center gap-1.5 rounded-full border px-4 text-[13px] font-semibold transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-teal-500";
    78→const pillIdle =
    79→  "border-sand-300 bg-white text-ink hover:border-teal-500/60 hover:bg-teal-50";

    81→
    82→const clamp01 = (v: number) => Math.min(100, Math.max(0, v));
    83→
    84→/* ── Composant principal ─────────────────────────────────────────────── */
    85→
    86→export function ProviderDirectory() {
    87→  const [q, setQ] = useState("");
    88→  const [type, setType] = useState<NonNullable<ProviderFilters["type"]>>("tous");
    89→  const [commune, setCommune] = useState("toutes");
    90→  const [open24, setOpen24] = useState(false);
    91→
    92→  const [results, setResults] = useState<Provider[]>([]);
    93→  const [loading, setLoading] = useState(true);
    94→
    95→  const [hoveredId, setHoveredId] = useState<string | null>(null);
    96→  const [selectedId, setSelectedId] = useState<string | null>(null);
    97→
    98→  const [detail, setDetail] = useState<Provider | null>(null);
    99→  const [drawerOpen, setDrawerOpen] = useState(false);
   100→
   101→  const [userPos, setUserPos] = useState<{ x: number; y: number } | null>(null);
   102→  const [locating, setLocating] = useState(false);
   103→  const [geoOpen, setGeoOpen] = useState(false);
   104→
   105→  const cardRefs = useRef(new Map<string, HTMLLIElement | null>());
   106→
   107→  // Appel mock `searchProviders` — la recherche texte est débouncée de 250 ms
   108→  // via un timer avec cleanup (les autres filtres s'appliquent immédiatement).
   109→  useEffect(() => {
   110→    let active = true;
   111→    const timer = setTimeout(
   112→      () => {
   113→        setLoading(true); // squelette pendant le fetch (saisie comme filtres)
   114→        searchProviders({ q: q.trim(), type, commune, open24 }).then((list) => {
   115→          if (!active) return;
   116→          setResults(list);
   117→          setLoading(false);
   118→        });
   119→      },
   120→      q.trim() ? 250 : 0
   121→    );
   122→    return () => {
   123→      active = false;
   124→      clearTimeout(timer);
   125→    };
   126→  }, [q, type, commune, open24]);
   127→
   128→  // Tri simulé « Autour de moi » : distance euclidienne sur le repère x/y de la carte.
   129→  const sorted = useMemo(() => {
   130→    if (!userPos) return results;
   131→    return [...results].sort(
   132→      (a, b) =>
   133→        Math.hypot(a.x - userPos.x, a.y - userPos.y) -
   134→        Math.hypot(b.x - userPos.x, b.y - userPos.y)
   135→    );
   136→  }, [results, userPos]);
   137→
   138→  const focusCard = (id: string) => {
   139→    setSelectedId(id);
   140→    cardRefs.current.get(id)?.scrollIntoView({ block: "nearest", behavior: "smooth" });
   141→  };
   142→
   143→  const openDetail = (p: Provider) => {
   144→    setDetail(p);
   145→    setDrawerOpen(true);
   146→  };
   147→
   148→  const activateGeo = () => {
   149→    setGeoOpen(false);
   150→    if (!("geolocation" in navigator)) {
   151→      toast.warning("La géolocalisation n’est pas disponible sur cet appareil.");
   152→      return;
   153→    }
   154→    setLocating(true);
   155→    navigator.geolocation.getCurrentPosition(
   156→      (pos) => {
   157→        // Projection simulée Bamako/Kati → repère 0–100 de la carte SVG.
   158→        const x = clamp01(((pos.coords.longitude + 8.1) / 0.22) * 100);
   159→        const y = clamp01(((12.82 - pos.coords.latitude) / 0.24) * 100);
   160→        setUserPos({ x, y });
   161→        setLocating(false);
   162→        toast.success("Position activée — prestataires triés par distance.");
   163→      },
   164→      () => {
   165→        setLocating(false);
   166→        toast.warning("Position refusée — l’annuaire reste trié par défaut.");
   167→      },
   168→      { timeout: 8000, maximumAge: 300000 }
   169→    );
   170→  };
   171→
   172→  const resetFilters = () => {
   173→    setQ("");
   174→    setType("tous");
   175→    setCommune("toutes");
   176→    setOpen24(false);
   177→    setUserPos(null);
   178→    setSelectedId(null);
   179→    setHoveredId(null);
   180→    setLoading(true);
   181→  };
   182→
   183→  return (
   184→    <section aria-label="Annuaire des prestataires conventionnés" className="py-14 sm:py-20">
   185→      <div className="container-ams">
   186→        {/* ── Filtres ─────────────────────────────────────────────── */}
   187→        <div className="rounded-[24px] border border-sand-200 bg-white p-5 shadow-soft sm:p-6">
   188→          <div className="grid gap-4 sm:grid-cols-[1fr_auto] sm:items-end">
   189→            <div className="space-y-1.5">
   190→              <Label htmlFor="recherche-prestataire">Rechercher un établissement</Label>
   191→              <div className="relative">
   192→                <Search
   193→                  className="pointer-events-none absolute left-3.5 top-1/2 size-4 -translate-y-1/2 text-muted-foreground"
   194→                  aria-hidden
   195→                />
   196→                <Input
   197→                  id="recherche-prestataire"
   198→                  type="search"
   199→                  value={q}
   200→                  onChange={(e) => {
   201→                    setQ(e.target.value);
   202→                    setLoading(true);
   203→                  }}
   204→                  placeholder="Nom, quartier, commune…"
   205→                  className="h-12 rounded-[10px] pl-10"
   206→                />
   207→              </div>
   208→            </div>
   209→
   210→            <Popover open={geoOpen} onOpenChange={setGeoOpen}>
   211→              <PopoverTrigger asChild>
   212→                <Button variant="outline" className="h-12 min-h-12" disabled={locating}>
   213→                  {locating ? (
   214→                    <Loader2 className="size-4 animate-spin" aria-hidden />
   215→                  ) : (
   216→                    <LocateFixed className="size-4" aria-hidden />
   217→                  )}
   218→                  {locating ? "Localisation…" : "Autour de moi"}
   219→                </Button>
   220→              </PopoverTrigger>
   221→              <PopoverContent align="end" className="w-72 rounded-2xl">
   222→                <p className="text-sm font-semibold text-foreground">Trier par distance</p>
   223→                <p className="mt-1 text-[13px] leading-relaxed text-muted-foreground">
   224→                  Nous utilisons votre position uniquement pour trier les prestataires par
   225→                  distance. Elle n’est jamais enregistrée ni partagée.
   226→                </p>
   227→                <Button className="mt-3 w-full" onClick={activateGeo}>
   228→                  Activer
   229→                </Button>
   230→              </PopoverContent>
   231→            </Popover>
   232→          </div>
   233→
   234→          <div className="mt-4 flex flex-wrap gap-2" role="group" aria-label="Filtrer par type d’établissement">
   235→            {providerTypes.map((t) => (
   236→              <button
   237→                key={t.id}
   238→                type="button"
   239→                aria-pressed={type === t.id}
   240→                onClick={() => {
   241→                  setType(t.id);
   242→                  setLoading(true);
   243→                }}
   244→                className={cn(pillBase, type === t.id ? pillActive : pillIdle)}
   245→              >
   246→                {t.label}
   247→              </button>
   248→            ))}
   249→          </div>
   250→
   251→          <div className="mt-2.5 flex flex-wrap gap-2" role="group" aria-label="Filtrer par commune">
   252→            {["toutes", ...communeList].map((c) => (
   253→              <button
   254→                key={c}
   255→                type="button"
   256→                aria-pressed={commune === c}
   257→                onClick={() => {
   258→                  setCommune(c);
   259→                  setLoading(true);
   260→                }}
   261→                className={cn(pillBase, commune === c ? pillActive : pillIdle)}
   262→              >
   263→                {c === "toutes" ? "Toutes les communes" : c}
   264→              </button>
   265→            ))}
   266→            <button
   267→              type="button"
   268→              aria-pressed={open24}
   269→              onClick={() => {
   270→                setOpen24((v) => !v);
   271→                setLoading(true);
   272→              }}
   273→              className={cn(pillBase, open24 ? pillActive : pillIdle)}
   274→            >
   275→              <Clock3 className="size-3.5" aria-hidden />
   276→              Ouvert 24 h/24
   277→            </button>
   278→          </div>
   279→        </div>
   280→
   281→        {/* ── Liste + carte ───────────────────────────────────────── */}
   282→        <div className="mt-8 grid gap-8 lg:grid-cols-2">
   283→          <div className="order-2 lg:order-1">
   284→            <div className="flex flex-wrap items-baseline justify-between gap-x-4 gap-y-1">
   285→              <h2 className="font-display text-xl font-bold text-ink">
   286→                {loading
   287→                  ? "Recherche…"
   288→                  : `${sorted.length} prestataire${sorted.length > 1 ? "s" : ""}`}
   289→              </h2>
   290→              <p className="text-[13px] text-muted-foreground" aria-live="polite">
   291→                {userPos ? "Triés par distance estimée" : `sur ${providers.length} conventionnés`}
   292→              </p>
   293→            </div>
   294→
   295→            {loading ? (
   296→              <ul className="mt-4 space-y-3" aria-hidden>
   297→                {[0, 1, 2].map((i) => (
   298→                  <li key={i} className="rounded-[16px] border border-sand-200 bg-white p-5 shadow-soft">
   299→                    <Skeleton className="h-5 w-2/3" />
   300→                    <Skeleton className="mt-3 h-4 w-1/2" />
   301→                    <Skeleton className="mt-2 h-4 w-1/3" />
   302→                  </li>
   303→                ))}
   304→              </ul>
   305→            ) : sorted.length === 0 ? (
   306→              <div className="mt-4 rounded-[16px] border border-dashed border-sand-300 bg-white/60 p-8 text-center">
   307→                <SearchX className="mx-auto size-8 text-muted-foreground" aria-hidden />
   308→                <p className="mt-3 font-display text-lg font-bold text-ink">
   309→                  Aucun prestataire trouvé
   310→                </p>
   311→                <p className="mt-1 text-[13.5px] text-muted-foreground">
   312→                  Essayez un autre nom, élargissez la zone ou retirez le filtre 24 h/24.
   313→                </p>
   314→                <Button variant="outline" className="mt-4" onClick={resetFilters}>
   315→                  Réinitialiser les filtres
   316→                </Button>
   317→              </div>
   318→            ) : (
   319→              <ul className="mt-4 space-y-3 lg:max-h-[640px] lg:overflow-y-auto lg:pr-1">
   320→                {sorted.map((p) => (
   321→                  <li
   322→                    key={p.id}
   323→                    ref={(el) => {
   324→                      cardRefs.current.set(p.id, el);
   325→                    }}
   326→                  >
   327→                    <button
   328→                      type="button"
   329→                      aria-haspopup="dialog"
   330→                      onMouseEnter={() => setHoveredId(p.id)}
   331→                      onMouseLeave={() => setHoveredId((cur) => (cur === p.id ? null : cur))}
   332→                      onFocus={() => setHoveredId(p.id)}
   333→                      onBlur={() => setHoveredId((cur) => (cur === p.id ? null : cur))}
   334→                      onClick={() => openDetail(p)}
   335→                      className={cn(
   336→                        "w-full rounded-[16px] border bg-white p-4 text-left shadow-soft transition-all hover:-translate-y-0.5 hover:shadow-lift focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-teal-500 sm:p-5",
   337→                        selectedId === p.id
   338→                          ? "border-teal-500 ring-2 ring-teal-500/30"
   339→                          : "border-sand-200"
   340→                      )}
   341→                    >
   342→                      <div className="flex items-start justify-between gap-3">
   343→                        <div className="min-w-0">
   344→                          <p className="truncate font-display text-[15.5px] font-bold text-ink">
   345→                            {p.name}
   346→                          </p>
   347→                          <p className="mt-0.5 text-[13px] text-muted-foreground">
   348→                            {typeLabels[p.type]} · {p.quartier}, {p.commune}
   349→                          </p>
   350→                        </div>
   351→                        <span
   352→                          className="mt-1 size-2.5 shrink-0 rounded-full"
   353→                          style={{ backgroundColor: typeColors[p.type] }}
   354→                          aria-hidden
   355→                        />
   356→                      </div>
   357→                      <div className="mt-2.5 flex flex-wrap items-center gap-x-3 gap-y-1 text-[12.5px] text-muted-foreground">
   358→                        <span className="inline-flex items-center gap-1.5">
   359→                          <Clock3 className="size-3.5" aria-hidden />
   360→                          {p.hours}
   361→                        </span>
   362→                        {p.open24 && (
   363→                          <Badge className="rounded-full bg-teal-100 text-[10.5px] font-bold text-teal-900">
   364→                            24 h/24
   365→                          </Badge>
   366→                        )}
   367→                      </div>
   368→                      <p className="mt-2.5 inline-flex items-center gap-1.5 text-[12.5px] font-semibold text-teal-700">
   369→                        Voir la fiche
   370→                        <ArrowRight className="size-3.5" aria-hidden />
   371→                      </p>
   372→                    </button>
   373→                  </li>
   374→                ))}
   375→              </ul>
   376→            )}
   377→          </div>
   378→
   379→          {/* ── Carte SVG stylisée ────────────────────────────────── */}
   380→          <div className="order-1 self-start lg:order-2 lg:sticky lg:top-24">
   381→            <div className="relative overflow-hidden rounded-[28px] border border-sand-200 bg-[linear-gradient(160deg,#F0FAF8,#F7F4EE)] p-4 shadow-soft">
   382→              <div className="absolute right-4 top-4 z-10 flex items-center gap-1.5 rounded-full bg-white/85 px-3 py-1.5 text-[11px] font-semibold text-ink shadow-xs backdrop-blur">
   383→                <Navigation className="size-3 text-teal-700" aria-hidden />
   384→                Bamako · Kati
   385→              </div>
   386→              {userPos && (
   387→                <div className="absolute left-4 top-4 z-10 rounded-full bg-ink/85 px-3 py-1.5 text-[11px] font-semibold text-sand-50 backdrop-blur">
   388→                  Tri par distance activé
   389→                </div>
   390→              )}
   391→
   392→              <svg
   393→                viewBox="0 0 100 100"
   394→                className={cn(
   395→                  "aspect-[4/3] w-full transition-opacity sm:aspect-square",
   396→                  loading && "opacity-50"
   397→                )}
   398→                role="img"
   399→                aria-label="Carte stylisée de Bamako — chaque point est un prestataire conventionné du résultat"
   400→              >
   401→                {/* fleuve Niger + axes + zones (motif repris de CareNetwork) */}
   402→                <g aria-hidden>
   403→                  <path
   404→                    d="M-2 34 C 15 30, 28 42, 42 38 S 66 26, 82 30 S 96 40, 104 38"
   405→                    fill="none"
   406→                    stroke="#7CCFC8"
   407→                    strokeWidth="5"
   408→                    strokeLinecap="round"
   409→                    opacity="0.55"
   410→                  />
   411→                  <path
   412→                    d="M-2 37 C 15 33, 28 45, 42 41 S 66 29, 82 33 S 96 43, 104 41"
   413→                    fill="none"
   414→                    stroke="#17A398"
   415→                    strokeWidth="1"
   416→                    strokeDasharray="2 3"
   417→                    opacity="0.5"
   418→                  />
   419→                  <g stroke="#EAE3D6" strokeWidth="1.6" strokeLinecap="round">
   420→                    <path d="M14 8 L 20 78 L 30 96" fill="none" />
   421→                    <path d="M8 60 L 48 52 L 88 62" fill="none" />
   422→                    <path d="M34 6 L 40 44 L 58 84" fill="none" />
   423→                    <path d="M62 10 L 60 40 L 70 90" fill="none" />
   424→                    <path d="M20 20 L 60 14 L 92 20" fill="none" />
   425→                  </g>
   426→                  <g fill="#0F5B57" opacity="0.05">
   427→                    <circle cx="20" cy="44" r="13" />
   428→                    <circle cx="38" cy="32" r="11" />
   429→                    <circle cx="55" cy="58" r="15" />
   430→                    <circle cx="70" cy="64" r="11" />
   431→                    <circle cx="12" cy="14" r="8" />
   432→                  </g>
   433→                </g>
   434→
   435→                {/* position utilisateur (géolocalisation simulée) */}
   436→                {userPos && (
   437→                  <g transform={`translate(${userPos.x} ${userPos.y})`}>
   438→                    <circle r="3.5" fill="none" stroke="#0B1F1E" strokeWidth="0.7">
   439→                      <animate
   440→                        attributeName="r"
   441→                        values="3;8;3"
   442→                        dur={`${duration.story}s`}
   443→                        repeatCount="indefinite"
   444→                      />
   445→                      <animate
   446→                        attributeName="opacity"
   447→                        values="0.8;0.1;0.8"
   448→                        dur={`${duration.story}s`}
   449→                        repeatCount="indefinite"
   450→                      />
   451→                    </circle>
   452→                    <circle r="2.2" fill="#0B1F1E" stroke="#F7F4EE" strokeWidth="0.7" />
   453→                  </g>
   454→                )}
   455→
   456→                {/* points prestataires synchronisés avec la liste */}
   457→                {sorted.map((p) => {
   458→                  const emphasized = hoveredId === p.id || selectedId === p.id;
   459→                  return (
   460→                    <g
   461→                      key={p.id}
   462→                      transform={`translate(${p.x} ${p.y})`}
   463→                      className="cursor-pointer focus:outline-none"
   464→                      tabIndex={0}
   465→                      role="button"
   466→                      aria-label={`${p.name}, ${p.commune} — afficher la fiche dans la liste`}
   467→                      onMouseEnter={() => setHoveredId(p.id)}
   468→                      onMouseLeave={() => setHoveredId((cur) => (cur === p.id ? null : cur))}
   469→                      onFocus={() => setHoveredId(p.id)}
   470→                      onBlur={() => setHoveredId((cur) => (cur === p.id ? null : cur))}
   471→                      onClick={() => focusCard(p.id)}
   472→                      onKeyDown={(e) => {
   473→                        if (e.key === "Enter" || e.key === " ") {
   474→                          e.preventDefault();
   475→                          focusCard(p.id);
   476→                        }
   477→                      }}
   478→                    >
   479→                      {emphasized && (
   480→                        <circle
   481→                          r="4"
   482→                          fill="none"
   483→                          stroke={typeColors[p.type] ?? "#17A398"}
   484→                          strokeWidth="0.9"
   485→                        >
   486→                          <animate
   487→                            attributeName="r"
   488→                            values="3;9.5;3"
   489→                            dur={`${duration.story}s`}
   490→                            repeatCount="indefinite"
   491→                          />
   492→                          <animate
   493→                            attributeName="opacity"
   494→                            values="0.95;0.15;0.95"
   495→                            dur={`${duration.story}s`}
   496→                            repeatCount="indefinite"
   497→                          />
   498→                        </circle>
   499→                      )}
   500→                      <circle
   501→                        r="3.2"
   502→                        fill={typeColors[p.type] ?? "#17A398"}
   503→                        opacity={emphasized ? 0.45 : 0.25}
   504→                      >
   505→                        <animate
   506→                          attributeName="r"
   507→                          values="3.2;6.5;3.2"
   508→                          dur={`${duration.base}s`}
   509→                          begin={`${(p.x % 1).toFixed(2)}s`}
   510→                          repeatCount="indefinite"
   511→                        />
   512→                        <animate
   513→                          attributeName="opacity"
   514→                          values="0.35;0;0.35"
   515→                          dur={`${duration.base}s`}
   516→                          begin={`${(p.x % 1).toFixed(2)}s`}
   517→                          repeatCount="indefinite"
   518→                        />
   519→                      </circle>
   520→                      <circle
